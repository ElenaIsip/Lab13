# Алгоритм и блок-схема
Начало. Передаются указатель на массив word и его максимальная длина max_len.

Инициализация: счетчик i = 0.

Пропуск пробелов:

Считать символ ch.

Если ch — пробел, табуляция или перевод строки, повторить считывание.

Если ch — конец файла (EOF), перейти к шагу 6.

Чтение слова:

Пока ch не EOF и не пробельный символ:

Если i < max_len - 1, записать ch в word[i], увеличить i.

Считать следующий символ ch.

Завершение строки: записать \0 в word[i].

Очистка буфера: пока ch не равен \n и не EOF, считывать и отбрасывать символы.

 # Реализация программы
    #include <stdio.h>
    #include <string.h>

    /* =========================================================
     ЗАДАНИЕ 1 (Английский текст на картинке)
      Чтение первого слова с помощью getchar()
          ========================================================= */
    void read_first_word(char *word, int max_len) {
    int ch;
    int i = 0;

    // 1. Пропускаем ведущие пробельные символы (пробел, табуляция, перевод строки)
    while ((ch = getchar()) != EOF) {
        if (ch != ' ' && ch != '\t' && ch != '\n') {
            break;
        }
    }

    // 2. Считываем слово до следующего пробельного символа
    while (ch != EOF && ch != ' ' && ch != '\t' && ch != '\n') {
        if (i < max_len - 1) { // оставляем место для \0
            word[i++] = (char)ch;
        }
        ch = getchar();
    }
    word[i] = '\0';

    // 3. Отбрасываем остаток строки до перевода строки
    while (ch != '\n' && ch != EOF) {
        ch = getchar();
    }
    }

    /* =========================================================
     ЗАДАНИЕ 2 (Индивидуальный вариант 22)
     Замена последовательных пробелов на один с помощью fgets()
     ========================================================= */
       void replace_multiple_spaces(void) {
       char str[256];
    char result[256];
    int j = 0;
    int in_space = 0;

    printf("Введите строку с лишними пробелами:\n");
    // Читаем строку, пока не встретим \n или конец файла
    if (fgets(str, sizeof(str), stdin) == NULL) {
        return; // Конец ввода
    }

    // Проходим по каждому символу
    for (int i = 0; str[i] != '\0'; i++) {
        if (str[i] == ' ') {
            if (!in_space) {
                result[j++] = ' '; // Первый пробел в серии
                in_space = 1;
            }
            // Остальные пробелы пропускаем
        } else {
            result[j++] = str[i]; // Обычный символ
            in_space = 0;
        }
    }
    result[j] = '\0'; // Завершаем строку

    printf("\nРезультат (лишние пробелы удалены):\n");
    printf("%s", result);
    }
